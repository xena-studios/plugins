# What scripts/boot-fabric.sh checks on each Minecraft line with xBackup, from the console: a
# backup with the pause and flush (noSave, then saveAllChunks with flush), a changed chunk read
# after the flush, a check, a staged restore and a clean-up, a reload, debug. Its folder is
# config/xbackup/, and every reply is drawn from lang/en.yml.
logged: xBackup 2.0.0 on
run: xbackup info
expect: xBackup 2.0.0 on

run: backup now boot test
expect: Backed up
run: backup list
expect: backups,
run: backup status
expect: Problem: none

# The flush really writes a changed chunk before the backup reads it.
run: forceload add 400 400 415 415
expect: Marked
run: backup now baseline
expect: Backed up
run: setblock 405 70 405 minecraft:gold_block
expect: Changed the block
run: backup now after a change
expect: changed chunk

run: backup verify full
expect: All good.

# Remote passwords live in secrets.yml, which is never backed up; config.yml is.
run: backup restore latest config/xbackup/secrets.yml
expect: has no config/xbackup/secrets.yml
run: backup restore latest server.properties
expect: would change

# A restore is planned first, staged on confirm, and can be called off.
run: backup restore latest server.properties confirm
expect: Staged
run: backup status
expect: Restore waiting
run: backup restore cancel
expect: Cancelled

run: backup prune
expect: Removed
run: xbackup reload
expect: Reloaded xBackup
run: backup now after a reload
expect: Backed up

run: xbackup debug
expect: Debug report saved as debug/debug-
