# xRewind on a real server, from the console.
# The database opens with the SQLite driver Paper and Folia ship, from a paper-plugin.
logged: Storage ready: the SQLite database history.db, schema 1.
run: xrewind info
expect: xRewind 0.1.0 on

run: xrewind reload
expect: Reloaded xRewind in

# Subcommands beside reload and info, and the writer ready for the database.
run: xr status
expect: Problem: none

# A lookup runs its SQL against the real database and replies.
run: xr lookup w:world t:1h
expect: Nothing found.
run: xr lookup t:1h
expect: From the console, add w:<world> or r:#global.

# Rollback's checks, and its SQL on a real database.
run: xr rollback u:Steve
expect: Add a time
run: xr rollback t:1h
expect: From the console, use r:#world with w:<world>, or r:#global.
run: xr rollback t:1h r:#global
expect: Nothing to change.
run: xr undo
expect: Nothing to undo.

# The whole loop on a real server: TNT blows up a block of planks, and a rollback
# puts it back. The reloads let the server tick between steps. Folia's console can't
# test blocks with execute, so it checks only what xRewind says.
run: forceload add 96 96 111 111
expect: Marked
run: fill 100 70 100 104 74 104 minecraft:oak_planks
expect: Successfully filled
run: setblock 102 71 103 minecraft:chest[facing=north,type=left]{Items:[{Slot:0b,id:"minecraft:diamond",count:5}]}
expect: Changed the block
run: setblock 101 71 103 minecraft:chest[facing=north,type=right]{Items:[{Slot:0b,id:"minecraft:emerald",count:3}]}
expect: Changed the block
run: summon minecraft:tnt 102 72 102 {fuse:0}
expect: Summoned new
run: xr reload
expect: Reloaded xRewind
run: xr reload
expect: Reloaded xRewind
paper run: execute if block 102 72 102 minecraft:air run say BLOWN UP
paper expect: BLOWN UP
run: xr lookup u:#tnt r:#global #count
expect: Lookup:
run: xr rollback u:#tnt t:1h r:#global
expect: Rolled back
paper run: execute if block 102 72 102 minecraft:oak_planks run say PUT BACK
paper expect: PUT BACK
paper run: execute if items block 102 71 103 container.* minecraft:diamond run say CHEST FULL
paper expect: CHEST FULL
# Each half of a double chest comes back with its own items.
paper run: execute if items block 101 71 103 container.* minecraft:emerald run say OTHER HALF FULL
paper expect: OTHER HALF FULL
run: xr undo
expect: Undid
paper run: execute if block 102 72 102 minecraft:air run say UNDONE
paper expect: UNDONE
