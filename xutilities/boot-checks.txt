# xUtilities on a real server, from the console.
run: xutilities info
expect: xUtilities 2.0.0 on

# Ours replaces vanilla /clear and /gamemode (vanilla says "No player was found").
run: clear Nobody
expect: No player named Nobody is online.
run: gamemode creative Nobody
expect: No player named Nobody is online.
run: discord
expect: Join our Discord: discord.gg/example

# Reloading re-registers commands while the server runs.
run: xu reload
expect: Reloaded xUtilities in
run: clear Nobody
expect: No player named Nobody is online.

# A datapack reload rebuilds the server's command tree; ours must survive it.
# It's done when it logs "Loaded N advancements". Folia has no /reload.
paper run: minecraft:reload
paper expect: advancements
paper run: clear Nobody
paper expect: No player named Nobody is online.
