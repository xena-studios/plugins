# xUtilities on a real Bukkit-family server, from the console (scripts/boot-test.sh --checks,
# through boot.sh): every command family, answered by the core's dispatcher and drawn from
# lang/en.yml.
logged: xUtilities 3.0.0 on
run: xutilities info
expect: xUtilities 3.0.0 on
run: xu
expect: commands on

# Ours replaces vanilla /clear and /gamemode, which say "No player was found".
run: clear Nobody
expect: No player called Nobody is online.
run: gamemode creative Nobody
expect: No player called Nobody is online.
run: creative Nobody
expect: No player called Nobody is online.
run: gm 9
expect: isn't one of survival, creative, adventure, spectator.

# Reloading re-registers the commands while the server runs: /back's listener is the one module.

run: xutilities:heal Nobody
