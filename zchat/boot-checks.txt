# zChat on a real server, from the console.
run: zchat info
expect: zChat 3.0.0 on

# Ours replaces Minecraft's /msg (vanilla says "No player was found").
run: msg Nobody hi
expect: No player named Nobody is online.
run: tell Nobody hi
expect: No player named Nobody is online.

run: mutechat
expect: Chat has been muted by
run: mutechat

run: clearchat

# Reloading re-registers commands while the server runs.
run: zc reload
expect: Reloaded zChat in
run: msg Nobody hi
expect: No player named Nobody is online.
